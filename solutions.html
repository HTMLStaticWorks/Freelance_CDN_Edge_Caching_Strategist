  const url = new URL(request.url)
  // Custom routing rule at the edge
  if (url.pathname.startsWith('/api/v2/')) {
    return fetch(request, {
      cf: { cacheTtl: 300, cacheEverything: true }
    })
  }
  return fetch(request)
}
